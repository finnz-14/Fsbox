FSBox DevKit - Mod SDK Reference

Complete SDK documentation. Every API with signature, use case, and runnable example.

Types

Color

type Color struct {
    R uint8 `json:"r"`
    G uint8 `json:"g"`
    B uint8 `json:"b"`
    A uint8 `json:"a"`
}

RGBA color value. Alpha channel ranges from 0 (transparent) to 255 (opaque).

// White opaque color
c := fsbox.Color{R: 255, G: 255, B: 255, A: 255}

// Half-transparent red
red := fsbox.Color{R: 255, G: 0, B: 0, A: 128}

Vec3

type Vec3 struct {
    X float32 `json:"x"`
    Y float32 `json:"y"`
    Z float32 `json:"z"`
}

3D position or vector in world space.

pos := fsbox.Vec3{X: 0, Y: 2, Z: 12}

Lifecycle

On(event, handler)On(string, func(fsbox.Event))Subscribe to engine events. Example events: "world.reset", "ui.click".

Examples

fsbox.On("world.reset", func(e fsbox.Event) {
    // world was reset; clear cached ids

    // runs every frame

World-space Billboards

Billboards are world-space panels that always face the camera. Use them for floating labels, health bars, or in-world menus beside entities.

Examples

panel, _ := fsbox.AddBillboardPanel(0, 2, 0, 2.5, 1.5)
btn, _ := fsbox.AddBillboardButton(panel, -1, -0.5, 0.8, 0.25, "PLAY", fsbox.Color{R: 255, G: 255, B: 255, A: 255})
_, _ = fsbox.AddBillboardTextWorld(panel, 0, 0.8, "Jenny", 0.15, fsbox.Color{R: 255, G: 255, B: 255, A: 255})

// update button text later
var newLabel string = "STOP"
fsbox.UpdateBillboardElement(btn, fsbox.BillboardElementUpdate{Label: &newLabel})

// hover detection
hover, _ := fsbox.GetBillboardHover()
if hover.Panel == panel { /* crosshair over our panel */ }

// move panel
fsbox.MoveBillboardPanel(panel, 3, 2, 5)
fsbox.RemoveBillboardPanel(panel)
fsbox.ClearBillboardPanels()

Full Examples

1. Spawn crate + play sound on key press

package main

import (
    fsbox "fsbox.dev/sdk"
)

func main() {
    fsbox.OnTick(func() {
        pressed, _ := fsbox.InputKeyPressed(fsbox.KeyE)
        if !pressed { return }
        info, _ := fsbox.GetPlayerInfo()
        pos := fsbox.Vec3{X: info.Position.X, Y: info.Position.Y + 1, Z: info.Position.Z}
        id, err := fsbox.SpawnPreset("Crate", pos, fsbox.Color{R: 200, G: 200, B: 200, A: 255})
        if err != nil { fsbox.Log(err.Error()); return }
        _ = fsbox.LoadSound("spawn", "spawn.wav")
        _ = fsbox.PlaySound("spawn")
        _ = id
    })
}

2. Floating billboard panel with PLAY/STOP

package main

import (
    "encoding/json"
    fsbox "fsbox.dev/sdk"
)

var entID int32
var playing int32 = -1

func main() {
    fsbox.On("ui.click", func(e fsbox.Event) {
        var d struct{ Kind string `json:"kind"`; ID int32 `json:"id"` }
        _ = json.Unmarshal(e.Data, &d)
        if d.Kind != "billboard_button" { return }
        if d.ID == playBtn { _ = fsbox.PlayAnimationByIndex(entID, 0, 1.0); playing = 0 }
        if d.ID == stopBtn { _ = fsbox.StopAnimation(entID); playing = -1 }
    })

    fsbox.OnTick(func() {
        if entID == 0 {
            info, _ := fsbox.GetPlayerInfo()
            pos := fsbox.Vec3{X: info.Position.X, Y: info.Position.Y + 1, Z: info.Position.Z}
            id, err := fsbox.SpawnPreset("Jenny", pos, fsbox.Color{R: 255, G: 255, B: 255, A: 255})
            if err == nil { entID = id }
        }
    })

    panel, _ := fsbox.AddBillboardPanel(3, 2, 0, 2.5, 1.2)
    playBtn, _ = fsbox.AddBillboardButton(panel, -1, -0.4, 0.7, 0.25, "PLAY", fsbox.Color{R: 70, G: 130, B: 74, A: 245})
    stopBtn, _ = fsbox.AddBillboardButton(panel, 1, -0.4, 0.7, 0.25, "STOP", fsbox.Color{R: 168, G: 62, B: 62, A: 245})
    _ = fsbox.AddBillboardTextWorld(panel, 0, 0.7, "ANIMATIONS", 0.12, fsbox.Color{R: 255, G: 255, B: 255, A: 255})
}
