FSBox DevKit - Mod SDK Reference

Complete SDK documentation. Every API with signature, use case, and runnable example.

Types

Color

type Color struct {
    R uint8 `json:"r"`
    G uint8 `json:"g"`
    B uint8 `json:"b"`
    A uint8 `json:"a"`
}

RGBA color value. Alpha channel ranges from 0 (transparent) to 255 (opaque).

// White opaque color
c := fsbox.Color{R: 255, G: 255, B: 255, A: 255}

// Half-transparent red
red := fsbox.Color{R: 255, G: 0, B: 0, A: 128}

Vec3

type Vec3 struct {
    X float32 `json:"x"`
    Y float32 `json:"y"`
    Z float32 `json:"z"`
}

3D position or vector in world space.

pos := fsbox.Vec3{X: 0, Y: 2, Z: 12}

Lifecycle

APISignatureUse case
On(event, handler)On(string, func(fsbox.Event))Subscribe to engine events. Example events: "world.reset", "ui.click".
OnTick(handler)OnTick(func())Register a function called every simulation frame.
Call(cmd, args)Call(string, any) (json.RawMessage, error)Low-level engine command router. Use only when no wrapper exists.
Log(msg)Log(string)Write a message to the engine console/log.
Toast(msg, c)Toast(string, fsbox.Color)Show a temporary on-screen toast notification.

Examples

fsbox.On("world.reset", func(e fsbox.Event) {
    // world was reset; clear cached ids
})
fsbox.OnTick(func() {
    // runs every frame
})
fsbox.Log("mod started")
fsbox.Toast("hello", fsbox.Color{R: 255, G: 255, B: 255, A: 255})
resp, err := fsbox.Call("engine.info", nil)

info, err := fsbox.GetEngineInfo()
if err != nil { fsbox.Log(err.Error()); return }

player := info.Player
_ = player.Position
_ = player.Forward

// Slow motion while a key is held
if down, _ := fsbox.InputKeyDown(fsbox.KeyLeftShift); down {
    _ = fsbox.SetTimeScale(0.25)
} else {
    _ = fsbox.SetTimeScale(1)
}

fsbox.BindKey(fsbox.KeyF5, "mod.save")
fsbox.On("mod.save", func(e fsbox.Event) { _ = fsbox.SaveGame() })

Networking

The dedicated server is authoritative. Mods run on the server and can inspect the roster, message clients, and forward custom events. On a connected client the same helpers expose the local view and let client mods send events to the server. In single-player NetStatus().Mode is "off".

APISignatureUse case
NetStatus()NetStatus() (NetStatusInfo, error)Role, port/address, map, player count and server tick.
NetPlayers()NetPlayers() ([]NetPlayerInfo, error)Roster with positions, look angles, fly state and held prop ID.
BroadcastToast(text, color)BroadcastToast(string, *Color) errorShow a HUD toast on every client. Server only.
SendNetEvent(event, data)SendNetEvent(string, any) errorSend a custom event to mods on the other side of the link.
KickPlayer(id, reason)KickPlayer(int32, string) (bool, error)Disconnect one client. Server only.

NetStatusInfo

type NetStatusInfo struct {
    Mode       string // "off", "server" or "client"
    Server     bool
    Connected  bool
    Address    string // client: remote host:port
    Port       int    // server: listen port
    Map        string
    Players    int
    PlayerID   int32
    PlayerName string
    Tick       uint64 // server simulation tick
    Protocol   int
}

NetPlayerInfo

type NetPlayerInfo struct {
    ID       int32
    Name     string
    Position Vec3
    Yaw      float32
    Pitch    float32
    Fly      bool
    Held     int32 // prop network ID, or 0
    Local    bool  // true for the client's own entry
}
BroadcastToast and KickPlayer only work on a dedicated server. SendNetEvent works in both directions: a server mod broadcasts to all clients, a client mod sends to the server. Clients receive events through their normal event handler.

Examples

// Server: welcome players and report the roster
status, _ := fsbox.NetStatus()
if status.Server {
    fsbox.Log("serving " + status.Map + " on port " + fmt.Sprint(status.Port))
    players, _ := fsbox.NetPlayers()
    for _, p := range players {
        fsbox.Log("player " + p.Name)
    }
    _ = fsbox.BroadcastToast("WELCOME TO THE SERVER", nil)
}

// Server: notify every client that a round started
_ = fsbox.SendNetEvent("round.start", map[string]any{
    "round": 3,
    "mode":  "survival",
})

// Client: send an event to the server mods
_ = fsbox.SendNetEvent("client.ready", nil)

// Server: remove a disruptive client
if ok, err := fsbox.KickPlayer(4, "idle too long"); err == nil && ok {
    fsbox.Log("kicked")
}

1. Spawn crate + play sound on key press

package main

import (
    fsbox "fsbox.dev/sdk"
)

func main() {
    fsbox.OnTick(func() {
        pressed, _ := fsbox.InputKeyPressed(fsbox.KeyE)
        if !pressed { return }
        info, _ := fsbox.GetPlayerInfo()
        pos := fsbox.Vec3{X: info.Position.X, Y: info.Position.Y + 1, Z: info.Position.Z}
        id, err := fsbox.SpawnPreset("Crate", pos, fsbox.Color{R: 200, G: 200, B: 200, A: 255})
        if err != nil { fsbox.Log(err.Error()); return }
        _ = fsbox.LoadSound("spawn", "spawn.wav")
        _ = fsbox.PlaySound("spawn")
        _ = id
    })
}

2. Floating billboard panel that recolors a crate

package main

import (
    "encoding/json"
    fsbox "fsbox.dev/sdk"
)

var (
    entID  int32
    redBtn int32
    newBtn int32
)

func main() {
    fsbox.On("ui.click", func(e fsbox.Event) {
        var d struct {
            Kind string `json:"kind"`
            ID   int32  `json:"id"`
        }
        if json.Unmarshal(e.Data, &d) != nil || d.Kind != "billboard_button" {
            return
        }
        if d.ID == redBtn {
            _ = fsbox.SetEntityColor(entID, fsbox.Color{R: 220, G: 60, B: 60, A: 255})
        }
        if d.ID == newBtn {
            _ = fsbox.SetEntityColor(entID, fsbox.RandomColor())
        }
    })

    fsbox.OnTick(func() {
        if entID != 0 {
            return
        }
        p, _ := fsbox.GetPlayerInfo()
        pos := fsbox.Vec3{X: p.Position.X, Y: p.Position.Y + 1, Z: p.Position.Z}
        id, err := fsbox.SpawnPreset("Crate", pos, fsbox.Color{R: 200, G: 200, B: 200, A: 255})
        if err == nil {
            entID = id
        }
    })

    panel, _ := fsbox.AddBillboardPanel(3, 2, 0, 2.5, 1.2)
    redBtn, _ = fsbox.AddBillboardButtonStyled(panel, -0.8, -0.4, 0.7, 0.25, "RED",
        fsbox.Color{R: 168, G: 62, B: 62, A: 245},
        fsbox.Color{R: 255, G: 255, B: 255, A: 255}, 18)
    newBtn, _ = fsbox.AddBillboardButtonStyled(panel, 0.1, -0.4, 0.7, 0.25, "RANDOM",
        fsbox.Color{R: 70, G: 130, B: 74, A: 245},
        fsbox.Color{R: 255, G: 255, B: 255, A: 255}, 18)
    _ = fsbox.AddBillboardTextWorld(panel, 0, 0.7, "CRATE", 0.12, fsbox.Color{R: 255, G: 255, B: 255, A: 255})
}
